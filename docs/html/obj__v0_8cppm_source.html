259auto obj::read(const std::string& file_name) -> void {
260 std::ifstream file(file_name);
261 if (!file.is_open()) throw std::runtime_error("[ERR] Could not open file");
262
263 std::string buf{ std::istreambuf_iterator<char>(file), std::istreambuf_iterator<char>() };
264
265 const char* p = buf.data();
266 const char* end = p + buf.size();
267
268 auto skip_ws = [](const char* p, const char* end) noexcept -> const char* {
269 while (p < end && (*p == ' ' || *p == '\t')) ++p;
270 return p;
271 };
272
273 auto skip_line = [](const char* p, const char* end) noexcept -> const char* {
274 while (p < end && *p != '\n') ++p;
275 return p < end ? p + 1 : p;
276 };
277
278 auto read_float = [&](const char* p, float& out) noexcept -> const char* {
279 p = skip_ws(p, end);
280 auto [ptr, _] = std::from_chars(p, end, out);
281 return ptr;
282 };
283
284 auto read_face_token = [&](const char* p, _face_vertex& fv) noexcept -> const char* {
285 p = skip_ws(p, end);
286 auto [p1, _1] = std::from_chars(p, end, fv.v); p = p1;
287 if (p < end && *p == '/') {
288 ++p;
289 if (p < end && *p != '/') { auto [p2, _2] = std::from_chars(p, end, fv.vt); p = p2; }
290 if (p < end && *p == '/') { ++p; auto [p3, _3] = std::from_chars(p, end, fv.vn); p = p3; }
291 }
292 return p;
293 };
294
295 while (p < end) {
296 p = skip_ws(p, end);
297 if (p >= end) break;
298 if (*p == '#' || *p == '\n' || *p == '\r') { p = skip_line(p, end); continue; }
299
300 const char* tok = p;
301 while (p < end && *p != ' ' && *p != '\t' && *p != '\n' && *p != '\r') ++p;
302 const std::string_view kw{ tok, static_cast<std::size_t>(p - tok) };
303
304 if (kw == "v") {
305 _vertex v{};
306 p = read_float(p, v.x);
307 p = read_float(p, v.y);
308 p = read_float(p, v.z);
309 p = read_float(p, v.w);
310 v_.push_back(v);
311 } else if (kw == "vt") {
312 _texture_coordinate vt{};
313 p = read_float(p, vt.u);
314 p = read_float(p, vt.v);
315 p = read_float(p, vt.w);
316 vt_.push_back(vt);
317 } else if (kw == "vn") {
318 _vertex_normal vn{};
319 p = read_float(p, vn.x);
320 p = read_float(p, vn.y);
321 p = read_float(p, vn.z);
322 vn_.push_back(vn);
323 } else if (kw == "vp") {
324 _parameter_space_vertex vp{};
325 p = read_float(p, vp.u);
326 p = read_float(p, vp.v);
327 p = read_float(p, vp.w);
328 vp_.push_back(vp);
329 } else if (kw == "f") {
330 _face f{};
331 p = skip_ws(p, end);
332 while (p < end && *p != '\n' && *p != '\r') {
333 _face_vertex fv{};
334 p = read_face_token(p, fv);
335 f.push_back(fv);
336 p = skip_ws(p, end);
337 }
338 f_.push_back(std::move(f));
339 } else if (kw == "l") {
340 _line l{};
341 p = skip_ws(p, end);
342 while (p < end && *p != '\n' && *p != '\r') {
343 _line_vertex lv{};
344 auto [ptr, _] = std::from_chars(p, end, lv);
345 p = ptr;
346 l.push_back(lv);
347 p = skip_ws(p, end);
348 }
349 l_.push_back(std::move(l));
350 }
351
352 p = skip_line(p, end);
353 }
354}

356auto obj::format_output() const -> std::string {
357 std::string out;
358 out.reserve(4096);
359
360 for (auto& v : v_) {
361 if (v.w != 0.f) {
362 out += std::format("Vertex: {{ x: {}, y: {}, z: {}, w: {} }}\n", v.x, v.y, v.z, v.w);
363 } else {
364 out += std::format("Vertex: {{ x: {}, y: {}, z: {} }}\n", v.x, v.y, v.z);
365 }
366 }
367 for (auto& vt : vt_) {
368 if (vt.v != 0.f && vt.w != 0.f) {
369 out += std::format("Vertex Texture: {{ u: {}, v: {}, w: {} }}\n", vt.u, vt.v, vt.w);
370 } else if (vt.v != 0.f) {
371 out += std::format("Vertex Texture: {{ u: {}, v: {} }}\n", vt.u, vt.v);
372 } else {
373 out += std::format("Vertex Texture: {{ u: {} }}\n", vt.u);
374 }
375 }
376 for (auto& vn : vn_)
377 out += std::format("Vertex Normal: {{ x: {}, y: {}, z: {} }}\n", vn.x, vn.y, vn.z);
378
379 for (auto& vp : vp_) {
380 if (vp.v != 0.f && vp.w != 0.f) {
381 out += std::format("Parameter Space Vertex: {{ u: {}, v: {}, w: {} }}\n", vp.u, vp.v, vp.w);
382 } else if (vp.v != 0.f) {
383 out += std::format("Parameter Space Vertex: {{ u: {}, v: {} }}\n", vp.u, vp.v);
384 } else {
385 out += std::format("Parameter Space Vertex: {{ u: {} }}\n", vp.u);
386 }
387 }
388 for (auto& f : f_) {
389 out += "Face: {";
390 for (auto& fv : f) {
391 if (fv.vt != 0 && fv.vn != 0) {
392 out += std::format(" [{}, {}, {}]", fv.v, fv.vt, fv.vn);
393 } else if (fv.vt != 0) {
394 out += std::format(" [{}, {}]", fv.v, fv.vt);
395 } else {
396 out += std::format(" [{}]", fv.v);
397 }
398 }
399 out += " }\n";
400 }
401
402 for (auto& l : l_) {
403 out += "Line: { ";
404 for (auto& lv : l) out += std::format("{}, ", lv);
405 out += "}\n";
406 }
407
408 return out;
409}

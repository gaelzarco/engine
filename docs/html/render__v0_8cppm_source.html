125/*
126 * @brief Helper function that builds two edge vectors from a common origin @c a to compute 2D
127 * cross-product for triangle rasterization.
128 *
129 * @param a First triangle vertex point (screen-space pixels).
130 * @param b Second triangle vertex point (screen-space pixels).
131 * @param c Third triangle vertex point (screen-space pixels).
132 *
133 * @return Determinant value
134 */
135auto get_determinant(const point& a, const point& b, const point& c) -> int {
136 const point ab = {b.x - a.x, b.y - a.y};
137 const point ac = {c.x - a.x, c.y - a.y};
138
139 const int result = ab.y * ac.x - ab.x * ac.y;
140
141 return result;
142}

167export auto triangle(point a, point b, point c,std::vector<std::uint32_t>& buff,
168const rgb_color& color, const std::size_t& width, const std::size_t& height) -> void {
169 if (get_determinant(a, b, c) == 0) return;
170 if (get_determinant(a, b, c) < 0) std::swap(b, c);
171
172 // Create triangle bounding box
173 const int xmin = std::min({a.x, b.x, c.x});
174 const int ymin = std::min({a.y, b.y, c.y});
175
176 const int xmax = std::max({a.x, b.x, c.x});
177 const int ymax = std::max({a.y, b.y, c.y});
178
179 for (auto y{ymin}; y <= ymax; ++y) {
180 for (auto x{xmin}; x <= xmax; ++x) {
181 if (!in_bounds(x, y, width, height)) continue;
182
183 const point p{x, y};
184
185 const auto bounds_a = get_determinant(b, c, p);
186 const auto bounds_b = get_determinant(c, a, p);
187 const auto bounds_c = get_determinant(a, b, p);
188
189 if (bounds_a >= 0 && bounds_b >= 0 && bounds_c >= 0) {
190 buff[static_cast<std::size_t>(y) * width + static_cast<std::size_t>(x)] =
191 MFB_RGB(color._r, color._g, color._b);

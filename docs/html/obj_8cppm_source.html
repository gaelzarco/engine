273auto obj::read(const std::string& file_name) -> void {
274 std::ifstream file(file_name);
275 if (!file.is_open()) throw std::runtime_error("[ERR] Could not open file");
276
277 std::string buf{ std::istreambuf_iterator<char>(file),
278 std::istreambuf_iterator<char>() };
279
280 const char* p = buf.data();
281 const char* end = p + buf.size();
282
283 auto skip_ws = [](const char* p, const char* end) noexcept -> const char* {
284 while (p < end && (*p == ' ' || *p == '\t')) ++p;
285 return p;
286 };
287
288 auto skip_line = [](const char* p, const char* end) noexcept ->
289 const char* {
290 while (p < end && *p != '\n') ++p;
291 return p < end ? p + 1 : p;
292 };
293
294 auto read_float = [&](const char* p, float& out) noexcept -> const char* {
295 p = skip_ws(p, end);
296 auto [ptr, _] = std::from_chars(p, end, out);
297 return ptr;
298 };
299
300 auto read_face_token = [&](const char* p, _face_vertex& fv) noexcept ->
301 const char* {
302 p = skip_ws(p, end);
303 auto [p1, _1] = std::from_chars(p, end, fv.v); p = p1;
304 if (p < end && *p == '/') {
305 ++p;
306 if (p < end && *p != '/') {
307 auto [p2, _2] = std::from_chars(p, end, fv.vt);
308 p = p2;
309 }
310 if (p < end && *p == '/') {
311 ++p; auto [p3, _3] = std::from_chars(p, end, fv.vn);
312 p = p3;
313
314 }
315 }
316 return p;
317 };
318
319 while (p < end) {
320 p = skip_ws(p, end);
321 if (p >= end) break;
322 if (*p == '#' || *p == '\n' || *p == '\r') {
323 p = skip_line(p, end);
324 continue;
325 }
326
327 const char* tok = p;
328 while (p < end && *p != ' ' && *p != '\t' && *p != '\n' && *p != '\r')
329 ++p;
330 const std::string_view kw{ tok, static_cast<std::size_t>(p - tok) };
331
332 if (kw == "v") {
333 _vertex v{};
334 p = read_float(p, v.x);
335 p = read_float(p, v.y);
336 p = read_float(p, v.z);
337 p = read_float(p, v.w);
338 v_.push_back(v);
339 } else if (kw == "vt") {
340 _texture_coordinate vt{};
341 p = read_float(p, vt.u);
342 p = read_float(p, vt.v);
343 p = read_float(p, vt.w);
344 vt_.push_back(vt);
345 } else if (kw == "vn") {
346 _vertex_normal vn{};
347 p = read_float(p, vn.x);
348 p = read_float(p, vn.y);
349 p = read_float(p, vn.z);
350 vn_.push_back(vn);
351 } else if (kw == "vp") {
352 _parameter_space_vertex vp{};
353 p = read_float(p, vp.u);
354 p = read_float(p, vp.v);
355 p = read_float(p, vp.w);
356 vp_.push_back(vp);
357 } else if (kw == "f") {
358 _face f{};
359 p = skip_ws(p, end);
360 while (p < end && *p != '\n' && *p != '\r') {
361 _face_vertex fv{};
362 p = read_face_token(p, fv);
363 f.push_back(fv);
364 p = skip_ws(p, end);
365 }
366 f_.push_back(std::move(f));
367 } else if (kw == "l") {
368 _line l{};
369 p = skip_ws(p, end);
370 while (p < end && *p != '\n' && *p != '\r') {
371 _line_vertex lv{};
372 auto [ptr, _] = std::from_chars(p, end, lv);
373 p = ptr;
374 l.push_back(lv);
375 p = skip_ws(p, end);
376 }
377 l_.push_back(std::move(l));
378 }
379
380 p = skip_line(p, end);
381 }
382}

388 for (auto& v : v_) {

392 out += std::format("Vertex: {{ x: {}, y: {}, z: {} }}\n", v.x, v.y, v.z);
393 }

398 vt.u, vt.v, vt.w);

404 }
405 }
406 for (auto& vn : vn_)

408 vn.x, vn.y, vn.z);
409

411 if (vp.v != 0.f && vp.w != 0.f) {
412 out += std::format("Parameter Space Vertex: {{ u: {}, v: {}, w: {} }}\n", vp.u, vp.v, vp.w);
413 } else if (vp.v != 0.f) {
414 out += std::format("Parameter Space Vertex: {{ u: {}, v: {} }}\n",
415 vp.u, vp.v);
416 } else {
417 out += std::format("Parameter Space Vertex: {{ u: {} }}\n", vp.u);
418 }
419 }
420 for (auto& f : f_) {
421 out += "Face: {";
422 for (auto& fv : f) {
423 if (fv.vt != 0 && fv.vn != 0) {
424 out += std::format(" [{}, {}, {}]", fv.v, fv.vt, fv.vn);

434 for (auto& l : l_) {
435 out += "Line: { ";

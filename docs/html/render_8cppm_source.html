148auto mfb_canvas::line(point a, point b, const rgb_color& color) -> void {
149 bool steep = std::abs(a.x - b.x) < std::abs(a.y - b.y);
150
151 if (steep) {
152 std::swap(a.x, a.y);
153 std::swap(b.x, b.y);
154 }
155
156 if (a.x > b.x) std::swap(a, b);
157
158 int y = a.y;
159 int err = 0;
160
161 for (int x{a.x}; x <= b.x; x++) {
162 int col = steep ? y : x;
163 int row = steep ? x : y;
164
165 if (!point_in_bounds(col, row)) {
166 std::println("[LOG] skipped point out of bounds");
167 } else {
168 _buffer[static_cast<std::size_t>(row) * _width +
169 static_cast<std::size_t>(col)] = MFB_RGB(color._r,
170 color._g, color._b);
171 }
172
173 err += 2 * std::abs(b.y - a.y);
174 if (err > b.x - a.x) {
175 y += b.y > a.y ? 1 : -1;
176 err -= 2 * (b.x - a.x);
177 }
178 }

181auto mfb_canvas::triangle(point a, point b, point c, const rgb_color& color) ->
182void {
183 if (get_determinant(a, b, c) == 0) return;
184 if (get_determinant(a, b, c) < 0) std::swap(b, c);
185
186 // Create triangle bounding box
187 const int xmin = std::min({a.x, b.x, c.x});
188 const int ymin = std::min({a.y, b.y, c.y});
189
190 const int xmax = std::max({a.x, b.x, c.x});
191 const int ymax = std::max({a.y, b.y, c.y});
192
193 for (auto y{ymin}; y <= ymax; ++y) {
194 for (auto x{xmin}; x <= xmax; ++x) {
195 if (!point_in_bounds(x, y)) continue;
196
197 const point p{x, y};
198
199 const auto bounds_a = get_determinant(b, c, p);
200 const auto bounds_b = get_determinant(c, a, p);
201 const auto bounds_c = get_determinant(a, b, p);
202
203 if (bounds_a >= 0 && bounds_b >= 0 && bounds_c >= 0) {
204 _buffer[static_cast<std::size_t>(y) * _width +
205 static_cast<std::size_t>(x)] = MFB_RGB(color._r, color._g,
206 color._b);
207 }
208 }
